# U25: the U12 lane model (proj4.py) with per-family latency factors for
# two concurrent Mac lanes. Usage: python3 u25model.py FAMILY=F ...
import sys
src = open('proj4.py').read().split("for c,lab in [")[0]
exec(src)
base_jobs_for = jobs_for
def jobs_for_f(mode, T, lf):
    J = base_jobs_for(mode, T, 1.0)
    return {n: (h, None if m is None else m * lf.get(fam(n), 1.0))
            for n, (h, m) in J.items()}
def run_f(lf, maclanes=2, scale=None):
    global S
    s0 = S
    if scale: S = scale
    w = assign(jobs_for_f('med', 12, lf), maclanes)
    out = [round(evaluate(jobs_for_f(m, 12, lf), w, maclanes), 1)
           for m in ['min', 'med', 'max']]
    nm = sum(1 for v in w.values() if v[0] == 'm')
    S = s0
    return out, nm
for arg in sys.argv[1:]:
    name, spec = arg.split(':', 1)
    lf = {}
    for kv in spec.split(','):
        k, v = kv.split('=')
        lf[k] = float(v)
    for lab, sc in (('21383', 1.0), ('41091', 41091/21383)):
        out, nm = run_f(lf, 2, sc)
        print(f"{name}\t{lab}\t{nm}\t" + "\t".join(map(str, out)))
