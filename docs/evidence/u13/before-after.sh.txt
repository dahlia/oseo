#!/usr/bin/env bash
# U13 Phase B before-and-after driver. Each repetition runs one arm's six
# runs and then the other's, so the two runs of a pair are minutes apart
# rather than adjacent: host drift is reduced, not controlled. It records the
# runner's own reported duration, tree CPU, wall time, and host load for each
# run, and measures an uninstrumented tree: the only inputs are the runner's
# report and `/usr/bin/time`.
#
# Run it from the repository root with the unit's commits at HEAD. It checks
# every source file the unit changes under packages/ and tools/ out of BASE
# for the before arm and out of HEAD for the after arm, rebuilds between
# arms because the runner loads several of those packages from their built
# output, and restores HEAD when it finishes.
#
#   BASE=<merge base> OUT=/tmp/u13/ba REPS=2 SHARD=1/100 ./before-after.sh
#
# The Oseo object and archive cache is redirected per arm and CPU count, so a
# cold run starts from an empty one. ZIG_GLOBAL_CACHE_DIR stays on the lane
# directory in every run and is never emptied.
set -euo pipefail

BASE=${BASE:?set BASE to the revision the change is measured against}
OUT=${OUT:?set OUT to an output directory}
REPS=${REPS:-2}
SHARD=${SHARD:-1/100}
CPU_COUNTS=${CPU_COUNTS:-"3 4"}
: "${ZIG_GLOBAL_CACHE_DIR:?set ZIG_GLOBAL_CACHE_DIR to the lane cache}"

mapfile -t FILES < <(git diff --name-only "$BASE" HEAD -- packages tools)
if [ "${#FILES[@]}" -eq 0 ]; then
  echo "no changed sources between $BASE and HEAD" >&2
  exit 1
fi

mkdir -p "$OUT"

select_arm() { # arm
  if [ "$1" = before ]; then
    git checkout "$BASE" -- "${FILES[@]}"
  else
    git checkout HEAD -- "${FILES[@]}"
  fi
  mise run build > "$OUT/build-$1.log" 2>&1
}

run() { # arm cpus cache-state rep
  local arm=$1 cpus=$2 state=$3 rep=$4
  local label="$arm-${cpus}cpu-$state-r$rep"
  local cache="$OUT/cache-$arm-$cpus"
  if [ "$state" = cold ]; then rm -rf "$cache"; fi
  mkdir -p "$cache"
  cut -d' ' -f1-3 /proc/loadavg > "$OUT/load-$label.txt"
  echo "== $label $(date -Is) load=$(cut -d' ' -f1 /proc/loadavg)"
  XDG_CACHE_HOME=$cache taskset -c "0-$((cpus - 1))" \
    /usr/bin/time -v -o "$OUT/time-$label.txt" \
    node tools/test262.ts --shard "$SHARD" > "$OUT/run-$label.log" 2>&1
  grep -o "duration=[0-9.]*ms pool=[0-9]*" "$OUT/run-$label.log"
}

for rep in $(seq 1 "$REPS"); do
  for arm in before after; do
    select_arm "$arm"
    for cpus in $CPU_COUNTS; do
      run "$arm" "$cpus" cold "$rep"
      run "$arm" "$cpus" warm "a$rep"
      run "$arm" "$cpus" warm "b$rep"
    done
  done
done
select_arm after
