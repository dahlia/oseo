import { writeFileSync, readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { cBackend } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/backend-c/src/index.ts';
import { compileSource, describeTarget } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/compiler/src/index.ts';
import { createNodeHost } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/host/src/index.ts';
import { babelFrontend } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/parser-babel/src/index.ts';
import { cRuntimeProvider } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/runtime-c/src/index.ts';
import { withNativeFixture } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/testkit/src/index.ts';
import { zigToolchain } from '/home/dahlia/orca/workspaces/oseo/perf-profiling/packages/toolchain-zig/src/index.ts';
const directory='/home/dahlia/.local/state/oseo-preview/lanes/perf-profiling/acceptance-20261109/long-smoke';
const source='function add(a, b) { return a + b; } let total = 0; for (let i = 0; i < 2000000; i++) { const o = { value: i % 7 }; total = add(total, o.value + 0.25); } console.log(total);';
writeFileSync(directory+'/smoke.js',source+'\n');
const compiled=compileSource(babelFrontend,{sourceId:'profiling-smoke.js',source},{specialization:'disabled'});
if(compiled.diagnostics.length || !compiled.mir) throw Error(JSON.stringify(compiled.diagnostics));
const observation=await withNativeFixture({backend:cBackend,host:createNodeHost(),input:compiled.mir,operation:'execute',runtime:cRuntimeProvider,target:describeTarget('linux-x86_64-gnu'),toolchain:zigToolchain,buildConfiguration:'optimized',keepArtifacts:true}, result=>result);
const bytes=readFileSync(observation.executablePath);
writeFileSync(directory+'/build-observation.json',JSON.stringify({...observation,emittedC:undefined,sha256:createHash('sha256').update(bytes).digest('hex'),bytes:bytes.length,role:'method-acceptance-smoke',timedArtifactId:null,source},null,2)+'\n');
writeFileSync(directory+'/generated.c',observation.emittedC);
console.log(JSON.stringify({executablePath:observation.executablePath,stdout:observation.stdout,exitStatus:observation.exitStatus}));
