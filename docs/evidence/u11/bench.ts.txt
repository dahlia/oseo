import assert from "node:assert/strict";
import { readFile, writeFile } from "node:fs/promises";
import {
  parseReviewedSubset, parseTest262Case, assembleTest262Source,
  suiteRoot, readHarnesses, nativeExecutor, createReviewedManifest,
  readSerializedManifest, canonicalizeManifestTarget,
} from "./u11-runner.ts";
import {
  parseReviewedManifest, serializeTest262Manifest,
} from "./test262-manifest.ts";
import { summarizeTest262 } from "../packages/testkit/src/test262-summary.ts";
const subset = parseReviewedSubset(
  await readFile("tests/test262/subset.yaml", "utf8"),
);
const root = await suiteRoot(subset.suiteRevision);
const harnesses = await readHarnesses(root);
if (process.env.U11_SAMPLE === "shard") {
  const selected = {
    ...subset,
    tests: subset.tests.filter((_, index) => index % 200 === 2),
  };
  const expected = await readSerializedManifest();
  const parts = new Map(expected.partitions.map((p) => [p.path, p.text]));
  const reviewed = parseReviewedManifest(
    expected.indexText, (p) => parts.get(p),
  );
  const paths = new Set(selected.tests.map((t) => t.path));
  const results = reviewed.results.filter((r) => paths.has(r.case.path));
  const run = await createReviewedManifest(selected, root, harnesses,
    undefined, { poolLimit: 1 });
  assert.deepEqual(
    serializeTest262Manifest(canonicalizeManifestTarget(run.manifest)),
    serializeTest262Manifest({ ...reviewed, results,
      summary: summarizeTest262(results) }),
  );
  await writeFile(process.env.U11_RESULT, JSON.stringify({
    metadata: run.metadata, summary: run.manifest.summary,
    paths: [...paths],
  }));
} else {
  const path = "test/built-ins/TypedArray/prototype/copyWithin/" +
    "coerced-values-end-detached.js";
  const source = await readFile(root + "/" + path, "utf8");
  const parsed = parseTest262Case(source, path, subset.suiteRevision);
  const start = performance.now();
  const result = await nativeExecutor.execute({
    mode: "script", sourceId: path, specialization: "enabled",
    source: assembleTest262Source(source, "non-strict", parsed.case,
      harnesses, false),
    fragment: {
      body: source, strict: false, raw: false,
      sources: [{ sourceId: "base.js", source: harnesses.base },
        ...parsed.case.includes.map((name) => ({ sourceId: name,
          source: harnesses.includes.get(name) }))],
    },
  });
  assert.equal(result.exitStatus, 0, JSON.stringify(result));
  await writeFile(process.env.U11_RESULT, JSON.stringify({
    path, strictness: "non-strict", specialization: "enabled",
    milliseconds: performance.now() - start, result,
  }));
}
